|Detection Script Content|<details><summary>Click to expand...</summary>$AVClient = 'Sophos Anti-Virus'<br>$AVProduct = Get-WmiObject -Namespace 'root\SecurityCenter2' -Class AntiVirusProduct \| Where-Object { $_.displayName -eq $AVClient } \| Select-Object -First 1<br>$AVSummary = New-Object -TypeName PSObject<br><br>If ($AVProduct) {<br>    $hexProductState = [Convert]::ToString($AVProduct.productState, 16).PadLeft(6, '0')<br>    $hexRealTimeProtection = $hexProductState.Substring(2, 2)<br>    $hexDefinitionStatus = $hexProductState.Substring(4, 2)<br><br>    $RealTimeProtectionStatus = switch ($hexRealTimeProtection) {<br>        '00' { 'Off' }<br>        '01' { 'Expired' }<br>        '10' { 'On' }<br>        '11' { 'Snoozed' }<br>        default { 'Unknown' }<br>    }<br><br>    $DefinitionStatus = switch ($hexDefinitionStatus) {<br>        '00' { 'Up to Date' }<br>        '10' { 'Out of Date' }<br>        default { 'Unknown' }<br>    }<br><br>    $AVSummary \| Add-Member -MemberType NoteProperty -Name "$AVClient" -Value $AVProduct.displayName<br>    $AVSummary \| Add-Member -MemberType NoteProperty -Name "$AVClient real time protection enabled" -Value $RealTimeProtectionStatus<br>    $AVSummary \| Add-Member -MemberType NoteProperty -Name "$AVClient definitions up-to-date" -Value $DefinitionStatus<br>}<br>Else {<br>    $AVSummary \| Add-Member -MemberType NoteProperty -Name "$AVClient" -Value 'Error: No Antivirus product found'<br>    $AVSummary \| Add-Member -MemberType NoteProperty -Name "$AVClient real time protection enabled" -Value 'Error: No Antivirus product found'<br>    $AVSummary \| Add-Member -MemberType NoteProperty -Name "$AVClient definitions up-to-date" -Value 'Error: No Antivirus product found'<br>}<br><br>return $AVSummary \| ConvertTo-Json -Compress<br></details>|
